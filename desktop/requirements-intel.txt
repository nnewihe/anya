# Build requirements for the INTEL (x86_64) desktop build.
#
# Everything here is pinned back from what requirements.txt uses on Apple
# silicon, and every pin has a reason. Do not bump one without checking the
# others — they are a set.
#
# THE ROOT CONSTRAINT: torch's last macOS x86_64 wheel is 2.2.2 (March 2024).
# Every torch release after that is arm64-only on macOS. So the Intel build is
# permanently on an older torch than the Apple silicon build, and everything
# else here follows from that.

# Last torch with a macOS x86_64 wheel, and its matching torchvision.
# ultralytics only declares torch>=1.8.0, so this is within support.
torch==2.2.2
torchvision==0.17.2

# torch 2.2.2 was built against the numpy 1.x C ABI. With numpy 2 installed it
# raises "module compiled with NumPy 1.x cannot be run in NumPy 2.x" on import.
numpy<2

# opencv >= 4.12 wheels require macOS 14, which excludes most of the Macs this
# build exists for. 4.11 is also the last line that resolves against numpy<2.
opencv-python<4.12

# Qt 6.5 is the last LTS supporting macOS 11 (Big Sur). Qt 6.8+ requires macOS
# 12, which would drop the oldest machines. PyQt6-Qt6 must match PyQt6 exactly.
PyQt6==6.5.3
PyQt6-Qt6==6.5.3

# Pinned to the same version the Apple silicon build uses: pipeline/models/
# yolo26n.pt was trained with ultralytics 8.3.222, so anything older cannot
# load it. Pure-Python wheel, so no architecture concerns.
ultralytics==8.4.36

# No architecture constraints on these — current versions have x86_64 wheels.
filterpy
joblib>=1.3
scikit-learn>=1.3
scipy>=1.10
pyinstaller>=6.0
